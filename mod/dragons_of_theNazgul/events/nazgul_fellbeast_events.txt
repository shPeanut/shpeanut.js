	type = character_event
	title = nazgul_fellbeast.0001.t
	desc = nazgul_fellbeast.0001.d
	theme = learning_learning_event # TODO: заменить на валидную тему CK3 1.19 / иконку из RIE
	
	left_portrait = {
		character = root
		animation = anger
	}
	right_portrait = {
		character = root	# TODO: когда будет модель дракона — показать её через dragon portrait
		animation = threat
	}
	
	immediate = {
		# спавн уже выполнен в nazgul_claim_fell_beast_effect;
		# здесь фиксируем привязку зверя к персонажу (механика AGOT)
		root = { set_dragon_rider = yes }	# TODO: точное имя эффекта из AGOT

		name = FLAVOR-nazgul_fell_beast_flavor.1
		add_dread = 50
		ai_chance = { base = 100 }
	}
}

# --- 0002: Гибель ездового зверя на войне ---
nazgul_fellbeast.0002 = {
	type = character_event
	title = nazgul_fellbeast.0002.t
	desc = nazgul_fellbeast.0002.d
	theme = warfare_conquest_event # TODO: заменить на валидную тему CK3 1.19 / иконку из RIE
	
	left_portrait = {
		character = root
		animation = grief
	}
	
	trigger = {
		has_trait = knows_fell_beast_riders
		is_alive = yes
	}
	
	immediate = {
		remove_trait = master_of_fell_beasts
	}
	
	option = {
		name = nazgul_fell_beast_flavor.3
		add_prestige = -100
		trigger_event = { id = nazgul_fellbeast.0001 days = 365 } # можно призвать нового
	}
}

# --- 0003: Дар Саурона (уникальный зверь для Лорда Назгул) ---
nazgul_fellbeast.0003 = {
	type = character_event
	title = nazgul_fellbeast.0003.t
	desc = nazgul_fellbeast.0003.d
	theme = roleplay_darkmagic_event # TODO: тема из RIE/ваниллы — сверить с events/common/event_themes
	
	left_portrait = {
		character = root
		animation = happiness
	}
	
	trigger = {
		has_trait = lord_of_the_nazgul	# TODO: ключ Кхота из RIE
		NOR = {
			has_character_flag = got_lord_of_angermont
		}
	}
	
	immediate = {
		add_character_flag = got_lord_of_angermont
		nazgul_claim_lord_of_angermont_effect = yes
	}
	
	option = {
		name = OK
		ai_chance = { base = 100 }
	}
}
