# ПРИМЕЧАНИЕ: проверка "кто может летать" вынесена в
# common/scripted_triggers/nazgul_fellbeast_triggers.txt

# Приручить падаль-зверя: спавн «дракона» AGOT + привязка наездника
nazgul_claim_fell_beast_effect = {
	save_event_target_as = fell_beast_claimant

	# --- 1. Спавн зверя как дракона AGOT ---
	create_character = {
		location = root.capital_province
		character_type = dragon
		template = nazgul_fell_beast_template                # см. common/character_templates/nazgul_fell_beasts.txt
		age = 60
		faith = root.faith
		culture = root.culture
		save_event_target_as = fell_beast
	}
	# [ALT] Если в scripted_effects AGOT есть нативный спавн, замени блок выше на:
	# create_dragon = { type = nazgul_dragon_fell_beast age = 60 }

	# --- 2. Привязка по конвенциям AGOT (owned_dragon/current_rider/dragonrider) ---
	nazgul_bind_fell_beast_effect = yes

	# --- 3. Трейт-мост (гарантирует работу если AGOT-события не подхватят пару) ---
	add_trait = knows_fell_beast_riders
	if = {
		limit = { has_trait = lord_of_the_nazgul }           # TODO(REVERSE): ключ Кхота из LoTR RIE
		add_trait = master_of_fell_beasts
	}

	# редкий шанс получить уникального Владыку Ангормаунта
	random_list = {
		90 = { }                                             # обычный fell beast
		10 = { trigger_event = { id = nazgul_fellbeast.0004 days = 30 random = 20 } }
	}

	# --- 4. Интеграция с системой Ужаса (Terror) из LoTR RIE ---
	# TODO(REVERSE): заменить на реальный эффект RIE (add/change terror)
	add_dread = 50
	add_prestige_level = 1

	custom_tooltip = nazgul_fell_beast_log_claims
}

# Особый экземпляр для Лорда Назгул (Кхот)
nazgul_claim_lord_of_angermont_effect = {

}
