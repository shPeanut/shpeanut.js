# Падающий зверь Назгула — базовый «дракон» (fell beast)
nazgul_dragon_fell_beast = {
	# --- Идентификация/графика ---
	# Реальный путь ассетов AGOT (Workshop ID 2962333032):
	# C:\Program Files (x86)\Steam\steamapps\workshop\content\1158310\2962333032\
	#   gfx\models\portraits\agot_creatures\dragon
	# В CK3 .mod-пути всегда начинаются с "gfx/" и используют прямые слеши.
	# AGOT регистрирует модели драконов через own .gfx-аттачменты, поэтому
	# сначала используем ИМЯ аттачмента, а не путь к файлу.
	graphicalize = yes
	portrait = agot_dragon					# TODO(подтвердить): имя gfx-аттачмента из AGOT
	model_folder = "gfx/models/portraits/agot_creatures/dragon"	# базовая папка моделей AGOT (реальный путь)
	# Если у beast'а будет своя перекрашенная модель в ЭТОМ моду, раскомментируй:
	# portrait = "gfx/models/portraits/agot_creatures/dragon/nazgul_fell_beast.mesh"

	# --- Жизненный цикл (как в AGOT) ---
	age = 0					# возраст при спавне
	lifespan = 600			# fell beasts почти бессмертны (в films гибнут только от всадника Рока)
	fertility = 0.0			# не размножаются естественно — выводятся тёмными ритуалами
	can_mate = no
	is_female = no			# в лоре пол fell beast неоднозначен; фиксируем как у AGOT-драконов

	# --- Привязка к езде (ключевая механика AGOT: dragon_riding) ---
	# В AGOT дракон становится ездовым через traits/effects; здесь
	# задаём минимальный «возраст крыла» и требуемый трейт наездника.
	minimum_riding_age = 40
	required_rider_trait = knows_fell_beast_riders	 # наш трейт-мост (см. traits ниже в README)
	rider_can_attack = yes
	rider_terror_add = 15	 # Nazgûl + fell beast = ужас (terrorsystem из RIE)

	# --- Боевые модификаторы host (аналог AGOT combat modifiers) ---
	host_damage = 15
	host_pursuit_speed = 20
	siege_devastation_speed = 0.25
	moves_at_the_speed_of_war = no

	# --- Прочее ---
	spawns_from_eggs = no	# fell beasts не несут яиц — появляются по событию summoning
	starts_as_egg = no
	can_be_killed_in_battle = yes
	death_by_flight_exhaustion = no
}
