	lotr_balan_lai_setup_effect = yes 

	### Confederation of Koronande
	
	if = {
		limit = {
			OR = {
				AND = {
					game_start_date >= 5000
					game_start_date <= 6000
				}
			}
		}
		title:c_korlan.holder = {
			create_confederation = {
				name = CONFEDERATION_NAME_KORONANDE
			}
			scope:new_confederation = {
				set_variable = {
					name = confederation_culture
					value = culture:koronandi
				}
				set_variable = {
					name = lotr_can_not_elevate_confederation
					value = yes
				}
			}
			every_independent_ruler = {
				limit = {
					is_landed = yes
					highest_held_title_tier > tier_barony
					highest_held_title_tier < tier_kingdom
					faith = faith:faith_koronande
					culture = culture:koronandi
					is_confederation_member = no
					any_held_title = {
						tier = tier_county
						OR = {
							kingdom = title:k_koronande
							any_neighboring_county = {
								OR = {
									kingdom = title:k_koronande
								}
							}
							county = title:c_osoran
						}
					}
				}
				scope:new_confederation = {
					add_confederation_member = prev
				}
			}
		}
	}

			add_unpressed_claim = title:d_balan_lai_leader

			}
		}
	}
