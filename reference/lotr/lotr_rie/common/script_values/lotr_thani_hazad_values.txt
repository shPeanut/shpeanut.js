total_yearly_interest = {
    value = 0
    if = {
        limit = { title:c_gidwama = { has_variable = debt_karasalas } }
        add = debt_karasalas_interest
    }
    if = {
        limit = { title:c_gidwama = { has_variable = debt_alsarias } }
        add = debt_alsarias_interest
    }
    if = {
        limit = { title:c_gidwama = { has_variable = debt_gimileth } }
        add = debt_gimileth_interest
    }
    if = {
        limit = { title:c_gidwama = { has_variable = debt_gimilthani } }
        add = debt_gimilthani_interest
    }
    if = {
        limit = { title:c_gidwama = { has_variable = debt_pel } }
        add = debt_pel_interest
    }
    if = {
        limit = { title:c_gidwama = { has_variable = debt_inzilloni } }
        add = debt_inzilloni_interest
    }
}

interest_to_income_yearly = {
    value = total_yearly_interest
    if = {
        limit = { yearly_character_income > 0 }
        divide = yearly_character_income
    }
}

hunt_no_activity_melee_danger_value = { # Used for the azrathani content
	value = 25
	root = {
		if = {
			limit = { exists = var:animal_type }
			if = {
				limit = {
					OR = {
						var:animal_type = flag:lion
						var:animal_type = flag:tiger
						var:animal_type = flag:leopard
						var:animal_type = flag:warg #LotR
						var:animal_type = flag:caragor #LotR
					}
				}
				add = 25
			}
			else_if = {
				limit = { var:animal_type = flag:bear }
				add = 20
			}
			else_if = {
				limit = {
					OR = {
						var:animal_type = flag:boar
						var:animal_type = flag:bison
						var:animal_type = flag:hart
						var:animal_type = flag:reindeer
						var:animal_type = flag:elk
					}
				}
				add = 15
			}
			else_if = {
				limit = {
					OR = {
						var:animal_type = flag:wolf
						var:animal_type = flag:hyena
						var:animal_type = flag:lynx
						var:animal_type = flag:stag
						var:animal_type = flag:antelope
					}
				}
				add = 10
			}
			else_if = {
				limit = {
					OR = {
						var:animal_type = flag:fox
						var:animal_type = flag:roe
						var:animal_type = flag:gazelle
					}
				}
				add = -10
			}
			else_if = {
				limit = { var:animal_type = flag:hare }
				add = -20
			}
		}
		# huntsmaster accolade bonuses
		if = {
			limit = {
				root = {
					any_active_accolade = {
						accolade_parameter = acclaimed_knight_reduces_hunting_danger
					}
				}
			}
			add = acclaimed_knight_hunt_danger_value
		}
		else_if = {
			limit = {
				root = {
					any_active_accolade = {
						accolade_parameter = acclaimed_knight_reduces_hunting_danger_high
					}
				}
			}
			add = acclaimed_knight_hunt_danger_value_high
		}
	}
}
