elven_situation_zero = 0
elven_fading_limit = 200
elven_waning_limit = 400
elven_enduring_limit = 600
elven_renewing_limit = 800
elven_thriving_limit = 1000

### ELVEN HOPE LEVEL CALCULATION ###

hope_level = {
	value = 0
	if = {
		limit = { is_good_undying = yes }
		
		if = { #Rulers
			limit = { 
				is_ruler = yes
				highest_held_title_tier >= tier_county
				NAND = {
					is_landless_adventurer = yes
					suzerain ?= { is_good_undying = yes }
				}
			}

			#Defining county_scope
			if = { #Landed rulers
				limit = { is_landed = yes }
				capital_county = { save_temporary_scope_as = county_scope }
			}
			else = { #Unlanded rulers
				domicile.domicile_location.county = { save_temporary_scope_as = county_scope }
			}

			#Calculating hope level
			if = { # Level 1 hope
				limit = {
					any_character_situation = {
						any_situation_sub_region = {
							sub_region_current_phase = situation_elven_fading
							situation_sub_region_has_county = scope:county_scope
						}
					}
				}
				add = 1
			}
			else_if = { # Level 2 hope
				limit = {
					any_character_situation = {
						any_situation_sub_region = {
							sub_region_current_phase = situation_elven_waning
							situation_sub_region_has_county = scope:county_scope
						}
					}
				}
				add = 2
			}
			else_if = { # Level 3 hope
				limit = {
					any_character_situation = {
						any_situation_sub_region = {
							sub_region_current_phase = situation_elven_enduring
							situation_sub_region_has_county = scope:county_scope
						}
					}
				}
				add = 3
			}
			else_if = { # Level 4 hope
				limit = {
					any_character_situation = {
						any_situation_sub_region = {
							sub_region_current_phase = situation_elven_renewing
							situation_sub_region_has_county = scope:county_scope
						}
					}
				}
				add = 4
			}
			else_if = { # Level 5 hope
				limit = {
					any_character_situation = {
						any_situation_sub_region = {
							sub_region_current_phase = situation_elven_thriving
							situation_sub_region_has_county = scope:county_scope
						}
					}
				}
				add = 5
			}
		}
		else_if = { #Vassal Rangers
			limit = {
				is_landless_adventurer = yes
				exists = suzerain
			}
			add = suzerain.hope_level
		}
		else_if = { #Barons
			limit = { is_ruler = yes }
			add = liege.hope_level
		}
		else_if = { #Courtiers
			limit = {
				employer ?= { is_good_undying = yes }
			}
			add = employer.hope_level
		}
		else_if = { #Guests
			limit = {
				host ?= { is_good_undying = yes }
			}
			add = host.hope_level
		}
		else = { #Wanderers - TBD
			add = 2
		}
	}
}
