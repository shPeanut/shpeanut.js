# Unite the Crowns of the Celduin
decision_celduin_empire = {
	title = decision_celduin_empire
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_castle_02.dds"
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 12
		empire = 12
		hegemony = 12
	}
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 100
	}
	selection_tooltip = decision_celduin_empire_tooltip
	desc = decision_celduin_empire_desc
	confirm_text = decision_celduin_empire_confirm
	decision_group_type = realms_in_exile_major
	
	sort_order = 800

	is_shown = {
		culture ?= { has_cultural_pillar = heritage_northron } # Available for any northron king of Dale or Dorwinion
		is_independent_ruler = yes
		OR = {
			has_title = title:k_dale
			top_liege = title:k_dale.holder
			has_title = title:k_dorwinion
			top_liege = title:k_dorwinion.holder
		}

		NOR = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:decision_dale_high_king
			}
		}
		NOR = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:decision_celduin_empire
			}
		}
	}

	is_valid = {
		OR = {
			has_title = title:k_dale
			has_title = title:k_dorwinion
		}
		prestige_level >= 5
		legitimacy_level >= 5
		piety_level >= 4
		is_independent_ruler = yes
		OR = {
			completely_controls = title:k_dale
			completely_controls = title:k_dorwinion
		}
		trigger_if = {
			limit = {
				has_title = title:k_dale
				NOT = { has_title = title:k_dorwinion }
			}
			has_relation_friend = title:k_dorwinion.holder
		}
		trigger_if = {
			limit = {
				has_title = title:k_dorwinion
				NOT = { has_title = title:k_dale }
			}
			has_relation_friend = title:k_dale.holder
		}
		trigger_if = {
			limit = {
				has_title = title:k_dale
				NOT = { has_title = title:k_dorwinion }
			}
			is_allied_to = title:k_dorwinion.holder
		}
		trigger_if = {
			limit = {
				has_title = title:k_dorwinion
				NOT = { has_title = title:k_dale }
			}
			is_allied_to = title:k_dale.holder
		}
		any_powerful_vassal = {
			count = all
			opinion = {
				target = root
				value >= 80
			}
		}
	}
	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_available_adult = yes
		is_at_war = no
	}

	cost = {
		gold = 500
		prestige = 1000
	}

	effect = {
		#Create the title and make it primary
		create_title_and_vassal_change = {
			type = created
			save_scope_as = title_change
			add_claim_on_loss = yes
		}
		title:e_celduin = {
			change_title_holder = {
				holder = root
				change = scope:title_change
			}
		}
		resolve_title_and_vassal_change = scope:title_change
		set_primary_title_to = title:e_celduin
		
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:decision_celduin_empire
		}
		
		root.dynasty = {
			add_dynasty_prestige = 750
		}
		
		title:k_dale = {
			set_de_jure_liege_title = title:e_celduin 
		}
		title:k_dorwinion = {
			set_de_jure_liege_title = title:e_celduin 
		}
		
		if = { 
			limit = { has_title = title:k_dale }
			custom_tooltip = {
				text = decision_celduin_empire_dale
				create_title_and_vassal_change = {
					type = swear_fealty
					save_scope_as = change
					add_claim_on_loss = no
				}
				title:k_dorwinion.holder = {
					change_liege = {
						liege = title:e_celduin.holder
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
		
		if = { 
			limit = { has_title = title:k_dorwinion }
			custom_tooltip = {
				text = decision_celduin_empire_dorwinion
				create_title_and_vassal_change = {
					type = swear_fealty
					save_scope_as = change
					add_claim_on_loss = no
				}
				title:k_dale.holder = {
					change_liege = {
						liege = title:e_celduin.holder
						change = scope:change
					}
				}
				resolve_title_and_vassal_change = scope:change
			}
		}
		
		custom_tooltip = {
			text = decision_dale_high_king_fealty
			every_neighboring_and_across_water_top_liege_realm_owner = {
				limit = {
					primary_title.tier < root.primary_title.tier
					culture ?= { has_cultural_pillar = heritage_northron }
				}
				add_to_list = dale_possible_vassals
			}
			create_title_and_vassal_change = {
				type = swear_fealty
				save_scope_as = change
				add_claim_on_loss = no
			}
			every_in_list = {
				list = dale_possible_vassals
				random = {
					chance = 40
					save_scope_as = new_dale_vassal
					change_liege = {
						liege = title:e_celduin.holder
						change = scope:change
					}
					title:e_celduin.holder = {
						send_interface_toast = {
							type = event_toast_effect_good
							title = decision_dale_high_king_fealty_tt
							left_icon = root
							right_icon = scope:new_dale_vassal
						}
					}
				}
			}
			resolve_title_and_vassal_change = scope:change
		}
		trigger_event = {
			id = dale.2005
			days = 1
		}
	}
}

# Create the League of the Winelands
decision_dorwinion_confederation = {
	title = decision_dorwinion_confederation
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_council_04.dds"
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 12
		duchy = 12
		kingdom = 0
		empire = 0
		hegemony = 0
	}
	ai_potential = {
		always = yes
	}
	ai_will_do = {
		base = 100
	}
	selection_tooltip = decision_dorwinion_confederation_tooltip
	desc = decision_dorwinion_confederation_desc
	confirm_text = decision_dorwinion_confederation_confirm
	decision_group_type = realms_in_exile_major
	
	sort_order = 700

	is_shown = {
		NOT = { exists = title:k_dorwinion.holder }
		is_human = yes
		is_independent_ruler = yes
		NOT = { faith_is_evil = yes }
		highest_held_title_tier <= tier_duchy
		capital_county.kingdom ?= title:k_dorwinion
		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:decision_dorwinion_confederation
			}
		}
		NOR = {
			has_character_flag = forming_confederation
			is_confederation_member = yes
		}
	}

	is_valid = {
		is_independent_ruler = yes
		highest_held_title_tier = tier_duchy
		is_tributary = no
		prestige_level >= 3
		legitimacy_level >= 3
		any_powerful_vassal = {
			count = all
			opinion = {
				target = root
				value >= 80
			}
		}
		custom_tooltip = {
			text = indep_rulers_in_dorwinion_influenced_tt
			lotr_dorwinion_influenced_rulers_count >= lotr_dorwinion_influenced_rulers_count_max
		}
	}
	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_available_adult = yes
		is_at_war = no
	}

	cost = {
		prestige = 1000
	}

	effect = {
		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:decision_dorwinion_confederation
		}
		title:k_dorwinion = {
    		every_de_jure_county_holder = {
				limit = {
					is_independent_ruler = yes
					is_landed = yes
					is_human = yes
					this != root
					highest_held_title_tier <= tier_duchy
					capital_county.kingdom ?= title:k_dorwinion
					OR = {
						opinion = {
							target = root
							value >= very_high_positive_opinion
						}
						root = { has_strong_hook = prev }
						root = { has_relation_friend = prev }
						root = { is_allied_to = prev }
					}
				}
				add_to_list = dorwinion_confederation_members
			}
		}
		root.dynasty = {
			add_dynasty_prestige = 250
		}
		custom_tooltip = {
			text = decision_dorwinion_confederation_tt
			create_confederation = {
				name = LEAGUE_OF_THE_WINELANDS
				#type = confederation_type:republican_leage - Consider making this a republican league after S&S
			}
		}
		# scope:new_confederation ?= {
			# set_variable = {
				# value = root.culture
			# }
		# }
		every_in_list = {
			list = dorwinion_confederation_members
			scope:new_confederation ?= {
				add_confederation_member = prev
			}
		}
		scope:new_confederation ?= {
			every_confederation_member = {
				trigger_event = {
					id = dorwinion.0501
				}
			}
		}
	}
}

# Abolish Elections #
decision_dorwinion_abolish_elections = {
	title = decision_dorwinion_abolish_elections
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_dorwinion_02.dds"
	}
	
	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 24
		kingdom = 24
		empire = 24
		hegemony = 24
	}
	ai_potential = {
		exists = primary_heir
		primary_heir = {
			dynasty = root.dynasty
		}
		is_at_war = no
		OR = {
			faith_is_evil = yes
			has_trait = arrogant
			has_trait = ambitious
			has_trait = callous
			has_trait = arbitrary
		}
		NOR = {
			has_trait = content
			has_trait = just
		}
	}
	ai_will_do = {
		base = 100
	}
	selection_tooltip = decision_dorwinion_abolish_elections_tool_tip
	desc = decision_dorwinion_abolish_elections_desc
	confirm_text = decision_dorwinion_abolish_elections_confirm
	decision_group_type = realms_in_exile_major
	
	is_shown = {
		is_ruler = yes
		is_landed = yes
		# any_held_title = {
		# 	tier = tier_county
		# 	kingdom = title:k_dorwinion
		# }
		OR = {
			has_title = title:k_dorwinion
			any_liege_or_above = {
				has_title = title:k_dorwinion
			}
		}
		title:k_dorwinion = {
			has_title_law = oligarchic_elective_succession_law
		}
		has_dynasty = yes
	}
	is_valid = {
		has_title = title:k_dorwinion # Only the King gets to change the rules
		custom_tooltip = {
			text = decision_dorwinion_abolish_elections_title_heir_is_your_heir
			title:k_dorwinion = {
				current_heir = root.player_heir
			}
		}
		trigger_if = {
			limit = { is_ai = yes }
			prestige_level >= high_prestige_level
		}
		trigger_else = {
			prestige_level >= very_high_prestige_level
		}
		has_realm_law = crown_authority_3
		legitimacy_level >= 3
		dynasty = { has_dynasty_modifier = winelords }
		OR = { # You're either loved or feared
			NOT = {
				any_powerful_vassal = {
					opinion = {
						target = root
						value < 50
					}
				}
			}
			trigger_if = {
				limit = { is_ai = yes }
				dread >= 10
			}
			trigger_else = {
				dread >= 50
			}
		}
	}
	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_at_war = no
	}
	effect = {
		title:k_dorwinion = {
			clear_title_laws = yes
		}
		if = {
			limit = { title:e_celduin = { has_title_law = oligarchic_elective_succession_law } }
			title:e_celduin = { clear_title_laws = yes }
		}
		add_prestige = 500
		#dynasty = { # Juke note: If anything, this should make you more infamous
		#	add_dynasty_prestige_level = 1
		#}
		add_tyranny = medium_tyranny_gain
		add_legitimacy = medium_legitimacy_loss

		trigger_event = {
			id = dorwinion.0004	# decision_dorwinion_abolish_elections
		}
	}
}
