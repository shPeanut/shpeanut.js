decision_wk_return_rings_sauron = {
	title = decision_wk_return_rings_sauron
	picture = {
		reference = "gfx/interface/illustrations/decisions/decision_mordor_02.dds"
	}
	selection_tooltip = decision_wk_return_rings_sauron_tooltip
	desc = decision_wk_return_rings_sauron_desc
	confirm_text = decision_wk_return_rings_sauron_confirm

	decision_group_type = realms_in_exile_major

	is_shown = {
		dynasty = dynasty:dynasty_witchking
		is_landed = yes
		has_character_flag = wk_got_dwarven_rings

		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:decision_wk_return_rings_sauron
			}
		}
	}

	is_valid = {
		dynasty = dynasty:dynasty_witchking
		sauron_is_alive = yes
		is_landed = yes
		has_character_flag = wk_got_dwarven_rings
		AND = {
			title:k_angmar.holder = {
				any_character_artifact = {
					has_variable = ring_of_broadbeams 
				}
			}
			title:k_angmar.holder = {
				any_character_artifact = {
					has_variable = ring_of_firebeards
				}
			}
		}
	}

	cost = {
		prestige = 1000
	}

	effect = {
		add_piety = massive_piety_gain

		title:c_amon_lanc.holder = {
			save_scope_as = sauron_necromancer
		}
		title:k_angmar.holder = {
			every_character_artifact = {
				limit = { has_variable = ring_of_dwarves }
				set_owner = {
					target = scope:sauron_necromancer
					history = {
						type = given
						actor = scope:root
						recipient = scope:sauron_necromancer
					}
				}
				if = {
					limit = { has_variable = ring_of_broadbeams }
					set_artifact_description =  ring_of_broadbeams_sauron_description
				}
				if = {
					limit = { has_variable = ring_of_firebeards }
					set_artifact_description =  ring_of_firebeards_sauron_description
				}
			}
		}

		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:decision_wk_return_rings_sauron
		}
	}

	ai_potential = {
		always = yes
	}
	ai_check_interval = 40
		
	ai_will_do = {
		base = 0
	}
}
