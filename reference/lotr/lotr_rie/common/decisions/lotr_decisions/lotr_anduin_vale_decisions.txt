# Become the All-thane

	picture = {

	}

	desc = decision_aladegana_desc
	confirm_text = decision_aladegana_confirm
	decision_group_type = realms_in_exile_major
	
	sort_order = 1000

	is_shown = {
		exists = capital_province
		capital_province = {
			OR = {
				geographical_region = middleearth_west_rhovanion_anduin
				geographical_region = middleearth_west_rhovanion_mirkwood
			}
		}

		OR = {
			culture = culture:woodmen
			culture = culture:valemen
			culture = culture:flodufulka
			culture = culture:gargath
            culture = { any_parent_culture_or_above = { this = culture:woodmen } }
            culture = { any_parent_culture_or_above = { this = culture:valemen } }
			culture = { any_parent_culture_or_above = { this = culture:flodufulka } }
			culture = { any_parent_culture_or_above = { this = culture:gargath } }
		}
		OR = {
			culture = { has_cultural_pillar = heritage_northron }
			culture = { has_cultural_pillar = heritage_igath }
		}
		is_independent_ruler = yes

		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:decision_aladegana
			}
		}

	}

	is_valid = {
		trigger_if = {
			limit = {
				is_ai = no
			}
			prestige_level >= very_high_prestige_level
			exists = title:d_west_bight
			OR = {
				has_title = title:d_west_bight
				any_vassal_or_below = {
					has_title = title:d_west_bight
				}
			}
			completely_controls = title:k_angaladh
			completely_controls = title:k_langwell
		}

		has_title = title:k_anduin_vale
		completely_controls = title:d_nivrim
		completely_controls = title:d_carrock_vale
		completely_controls = title:d_warnelath
		completely_controls = title:d_ronen_in_anduin
		OR = {
			completely_controls = title:d_men_i_naugrim
			completely_controls = title:d_angaladh_rhunen
		}

	}
	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_available_adult = yes
		is_at_war = no
	}
	cost = {
		prestige = 450
	}
	effect = {
		#Create the title and make it primary
		create_title_and_vassal_change = {
			type = created
			save_scope_as = title_change
			add_claim_on_loss = yes
		}
		title:e_anduin_vale = {
			change_title_holder = {
				holder = root
				change = scope:title_change
			}
		}
		resolve_title_and_vassal_change = scope:title_change
		set_primary_title_to = title:e_anduin_vale

		root.dynasty = {
			add_dynasty_modifier = aladegan
			add_dynasty_prestige = 500
		}

		title:e_anduin_vale = { set_title_name = e_aladegana }

		hidden_effect = {
			every_realm_de_jure_kingdom = {
				if = {
					limit = {
						OR = {
							kingdom = title:k_anduin_vale
							kingdom = title:k_langwell
							kingdom = title:k_angaladh
							# duchy = title:d_nivrim
						}
					}
					set_de_jure_liege_title = title:e_anduin_vale
				}
			}
			every_realm_de_jure_duchy = {
				if = {
					limit = {
						duchy = title:d_nivrim
					}
					set_de_jure_liege_title = title:k_angaladh
				}
			}
		}
		trigger_event = {
			id = anduin_vale.0102
			days = 1
		}

		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:decision_aladegana
		}

		if = {
			limit = { # Major News
				NOT = { has_game_rule = no_news }
			}
			every_player = { #Global News Announcement
				trigger_event = news_event.0018
			}
		}
	}

	ai_check_interval_by_tier = {
		barony = 0
		county = 0
		duchy = 0
		kingdom = 36
		empire = 24
		hegemony = 12
	}
}

		}

		NOT = {
			is_target_in_global_variable_list = {
				name = unavailable_unique_decisions
				target = flag:decision_secure_the_high_pass
			}
		}
		# NOT = {
			# title:d_high_pass.holder ?= { faith_is_good = yes }
		# }
	}

	# IS VALID
	is_valid = {
		custom_tooltip = {
			text = must_hold_one_county_high_pass
			title:d_high_pass = {
				any_de_jure_county = {
					OR = {
						holder = root
						holder = { is_vassal_or_below_of = root }
					}
				}
			}
		}
		OR = {
			title:c_goblin_town.title_province = { has_holding_type = wastelands_holding }
			title:c_goblin_town.holder.top_liege ?= { is_dwarf = yes }
		}
	}

	is_valid_showing_failures_only = {
		is_imprisoned = no
		is_available_adult = yes
		is_at_war = no
	}

	cost = {
		gold = 150
		prestige = 500
	}

	# EFFECT
	effect = {
		title:d_high_pass = {
			every_in_de_jure_hierarchy = { # Every pass county held by orcs, take it
				limit = {
					tier = tier_county
					holder = { is_orc = yes }
				}
				create_title_and_vassal_change = {
					type = conquest
					save_scope_as = change
					add_claim_on_loss = yes
				}
				change_title_holder = {
					holder = root
					change = scope:change
				}
				resolve_title_and_vassal_change = scope:change
			}
			every_in_de_jure_hierarchy = { # Every pass wastelands county, make a colony
				limit = {
					tier = tier_county
					holder = { is_wastelands = yes }
				}
				ai_colonisation_effect = { WASTELANDS = this.title_province ROOT_SCOPE = root }
			}
		}
		custom_tooltip = {
			text = get_high_pass_secured_modifier
			title:d_high_pass = {
				every_in_de_jure_hierarchy = {
					add_county_modifier = {
						modifier = secured_high_pass_modifier
						years = 25
					}
				}
			}
		}
		send_interface_toast = {
			type = event_toast_effect_good
			title = decision_secure_the_high_pass_tt
			left_icon = root
			root.dynasty = { add_dynasty_prestige = 300 }
			add_legitimacy = major_legitimacy_gain
		}

		add_to_global_variable_list = {
			name = unavailable_unique_decisions
			target = flag:decision_secure_the_high_pass
		}
	}
}
