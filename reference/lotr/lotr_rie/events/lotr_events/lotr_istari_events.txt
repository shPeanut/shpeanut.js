istari.0002 = { # Find the appropriate target for istari.0003
	hidden = yes

	trigger = {
		is_istari = yes
		faith_is_good = yes
	}

	immediate = {
		root = { save_scope_as = departing_istar }
		if = {
			limit = { exists = root.employer }
			root.employer = {
				trigger_event = {
					id = istari.0003
					days = 1
				}
			}
		}
		else_if = { 
			limit = { exists = root.host }
			root.host = {
				trigger_event = {
					id = istari.0003
					days = 1
				}
			}
		}
		else_if = { 
			limit = { exists = root.liege }
			root.liege = {
				trigger_event = {
					id = istari.0003
					days = 1
				}
			}
		} 
	}
}

### Blue Wizards Events ###
# .0100-.0109 - Utility events
# .0110-0.149 - ROA Steppe Path
# .0150-0.179 - ROA Sirayn Path
# .0180-0.199 - ROA Khy Path

istari.0100 = { # Choose who to play as
	content_source = realms_dlc
	type = character_event
	title = istari.0100.t
	desc = istari.0100.desc
	theme = travel
	left_portrait = {
		character = character:lineofpallando
		animation = personality_rational
	}
	right_portrait = {
		character = character:lineofalatar
		animation = personality_zealous
	}

	trigger = {
		character:lineofalatar = {
			is_alive = yes
		}
		character:lineofpallando = {
			is_alive = yes
		}
	}

	immediate = {
		title:d_laamp_blues.holder = {
			save_scope_as = current_leader
		}
		if = {
			limit = {
				scope:current_leader = character:lineofpallando
			}
			character:lineofalatar = {
				save_scope_as = other_wizard
			}
		}
		else = {
			character:lineofpallando = {
				save_scope_as = other_wizard
			}
		}
	}

	option = { # I'll stay with Pallando
		name = istari.0100.a
		scope:current_leader = {
			set_designated_heir_unsafe = scope:other_wizard
		}
		ai_chance = { 
			base = 75
		}
	}

	option = { # Swap to Alatar
		name = istari.0100.b
		scope:current_leader = {
			set_designated_heir_unsafe = scope:other_wizard
			depose = yes
			add_prestige = minor_prestige_gain
		}
		ai_chance = { 
			base = 25
		}
	}

	after = {
		if = {
			limit = { 
				scope:current_leader != title:d_laamp_blues.holder
			}
		}
		scope:current_leader = {
			set_employer = title:d_laamp_blues.holder
		}
		title:d_laamp_blues.holder = {
			try_start_diarchy = lord_blue_wizards_diarchy
			set_diarch = this.player_heir
		}
	}
}
